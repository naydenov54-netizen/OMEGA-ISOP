1. PURPOSE
============================================================

This record documents the first executed CAPD interval-ODE smoke
test in the GR-TRANSFER-06C toolchain evaluation.

Test problem:
  y'(t) = y(t)
  y(0) = 1
  t in [0,1]

Exact solution:
  y(t) = exp(t)
  y(1) = e

The purpose is to verify that the selected CAPD build can compile
and execute an interval ODE integration, return an interval state
at t=1, and provide an enclosure over the last integration step.

This is a toolchain smoke test only. It is not a validation of the
GR-TRANSFER-06C model or of its transfer-matrix root.

============================================================
4. EXECUTED TEST PROGRAM
============================================================

The compiled source was:

#include <iostream>
#include <iomanip>
#include "capd/capdlib.h"

using namespace capd;
using namespace std;

int main() {
    cout << setprecision(17);

    IMap ode("time:t;var:y;fun:y;");
    IOdeSolver solver(ode, 20);
    ITimeMap timeMap(solver);

    IVector u(1);
    u[0] = 1.0;

    interval initTime = 0.0;
    interval finalTime = 1.0;

    C0HORect2Set set(u, initTime);
    timeMap.stopAfterStep(true);

    unsigned steps = 0;

    do {
        timeMap(finalTime, set);
        ++steps;

        cout << "STEP " << steps << endl;
        cout << "currentTime = "
             << set.getCurrentTime() << endl;
        cout << "state enclosure = "
             << (IVector)set << endl;
        cout << "trajectory enclosure = "
             << set.getLastEnclosure() << endl;

        if (steps > 10000) {
            cerr << "FAIL: step limit exceeded" << endl;
            return 2;
        }
    } while (!timeMap.completed());

    cout << "FINAL STATE = " << (IVector)set << endl;
    cout << "FINAL TIME = " << set.getCurrentTime() << endl;
    cout << "STEP COUNT = " << steps << endl;
    cout << "ODE INTEGRATION COMPLETED" << endl;

    return 0;
}
